def dijkstra(G, s):

        M.remove(v)

### Kereses
Binaris kereses rendezett tombben $O(\log n)$
```python
def binary_search(arr, val):
  left, right = 0, len(arr) - 1
  while left <= right:
    middle = (left + right) // 2
	if arr[middle] == val:
	  return middle
	elif arr[middle] < val:
	  left = middle + 1
	else:
	  right = middle - 1
  return -1  # not found
```

def bfs(graph, start):
  visited = set()
  queue = deque([start])
  visited.add(start)
  while queue:
    node = queue.popleft()
	print(node, end=" ")
	for neighbor in graph[node]:

### DFS $O(\lvert V \rvert + \lvert E \rvert)$

def dfs(graph, node, visited=None):
  if visited is None:
    visited = set()
    visited.add(node)
    print(node, end=" ")
    for neighbor in graph[node]:
      if neighbor not in visited:
	    dfs(graph, neighbor, visited)
